# hive loop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a native `hive loop` command that drives the headless Claude Code CLI iteratively until a task is complete, seeded by MindHive memory and bounded by real token/cost/context safeguards.

**Architecture:** A new `src/mindhive/loop/` package. Pure modules (config, contract, budget, context, safeguards, task) are fully unit-tested with table-driven cases; `driver.py` is the only boundary to the real `claude` CLI and is faked in tests; `engine.py` wires the cycle and emits `loop_*` events through the existing `events` ledger. No new runtime dependencies (stdlib `tomllib`, `subprocess`, `shutil`).

**Tech Stack:** Python ≥3.11, pytest, stdlib only. Integrates with existing `mindhive.events`, `mindhive.briefing`, `mindhive.recall`, `mindhive.guard`, `mindhive.paths`.

## Conventions (read once)

- Tests live in `tests/test_loop_<module>.py`. Every test that touches the store sets:
  ```python
  monkeypatch.setenv("MINDHIVE_HOME", str(tmp_path))
  monkeypatch.setenv("MINDHIVE_PROJECT", "test")
  ```
- Run a single test: `python -m pytest tests/test_loop_<module>.py::<name> -v`
- Run all loop tests: `python -m pytest tests/ -k loop -q`
- All new event types must be registered in `events.VALID_TYPES` or `append_event` raises `ValueError`.
- Pure functions take `now` (epoch seconds, `float`) as an argument — never call `time.time()` inside them — so tests are deterministic.

# src/mindhive/loop/config.py
import tomllib
from dataclasses import dataclass
from pathlib import Path


@dataclass
class LoopConfig:
    # [model]
    context_window: int = 200000
    smart_zone_ratio: float = 0.5
    model: str | None = None
    effort: str | None = None
    # [budget]
    window_hours: float = 5.0
    window_token_budget: int = 0          # 0 = unlimited
    window_cost_budget_usd: float = 0.0   # 0 = unlimited
    on_cap: str = "pause"                 # "pause" | "stop"
    # [loop]
    max_iterations: int = 50
    max_hours: float = 8.0
    allowed_tools: str = "Write,Read,Edit,Bash(git *),Bash(npm *)"
    iteration_timeout_minutes: int = 15
    # [breaker]
    no_progress_threshold: int = 3
    same_error_threshold: int = 5
    cooldown_minutes: int = 30

    @property
    def cooldown_seconds(self) -> float:
        return self.cooldown_minutes * 60

    @property
    def iteration_timeout_seconds(self) -> float:
        return self.iteration_timeout_minutes * 60


def load_config(path: Path) -> LoopConfig:
    """Load loop.toml, falling back to defaults for any missing key/section."""
    cfg = LoopConfig()
    path = Path(path)
    if not path.exists():
        return cfg
    data = tomllib.loads(path.read_text(encoding="utf-8"))
    model = data.get("model", {})
    budget = data.get("budget", {})
    loop = data.get("loop", {})
    breaker = data.get("breaker", {})
    return LoopConfig(
        context_window=model.get("context_window", cfg.context_window),
        smart_zone_ratio=model.get("smart_zone_ratio", cfg.smart_zone_ratio),
        model=model.get("model", cfg.model),
        effort=model.get("effort", cfg.effort),
        window_hours=budget.get("window_hours", cfg.window_hours),
        window_token_budget=budget.get("window_token_budget", cfg.window_token_budget),
        window_cost_budget_usd=budget.get("window_cost_budget_usd", cfg.window_cost_budget_usd),
        on_cap=budget.get("on_cap", cfg.on_cap),
        max_iterations=loop.get("max_iterations", cfg.max_iterations),
        max_hours=loop.get("max_hours", cfg.max_hours),
        allowed_tools=loop.get("allowed_tools", cfg.allowed_tools),
        iteration_timeout_minutes=loop.get("iteration_timeout_minutes", cfg.iteration_timeout_minutes),
        no_progress_threshold=breaker.get("no_progress_threshold", cfg.no_progress_threshold),
        same_error_threshold=breaker.get("same_error_threshold", cfg.same_error_threshold),
        cooldown_minutes=breaker.get("cooldown_minutes", cfg.cooldown_minutes),
    )

# src/mindhive/loop/task.py
import re
from dataclasses import dataclass
from pathlib import Path

_FUTURE_HEADING = re.compile(r"^\s*#+\s*(future|optional)", re.IGNORECASE)
_UNCHECKED = re.compile(r"^\s*[-*]\s*\[\s\]")


@dataclass
class TaskFiles:
    prompt: str
    fix_plan: str
    agent: str


def _read(path: Path) -> str:
    return path.read_text(encoding="utf-8") if path.exists() else ""


def load_tasks(loop_dir: Path) -> TaskFiles:
    loop_dir = Path(loop_dir)
    return TaskFiles(
        prompt=_read(loop_dir / "PROMPT.md"),
        fix_plan=_read(loop_dir / "fix_plan.md"),
        agent=_read(loop_dir / "AGENT.md"),
    )


def required_remaining_count(fix_plan: str) -> int:
    """Count unchecked checklist items that appear BEFORE any Future/Optional
    heading. Items under such a heading are non-blocking and never counted."""
    count = 0
    for line in fix_plan.splitlines():
        if _FUTURE_HEADING.match(line):
            break
        if _UNCHECKED.match(line):
            count += 1
    return count
```

# src/mindhive/loop/contract.py
import re
from dataclasses import dataclass

_EXIT_RE = re.compile(r"EXIT_SIGNAL:\s*(true|false)", re.IGNORECASE)
_STATUS_RE = re.compile(r"STATUS:\s*(COMPLETE|IN_PROGRESS)", re.IGNORECASE)
_PROGRESS_RE = re.compile(r"progress:\s*(true|false)", re.IGNORECASE)
_SUMMARY_RE = re.compile(r"summary:\s*(.+)", re.IGNORECASE)


def _to_bool(m):
    return None if m is None else (m.group(1).lower() == "true")


@dataclass
class LoopStatus:
    summary: str | None
    progress: bool | None
    status: str | None            # "COMPLETE" | "IN_PROGRESS" | None
    exit_signal: bool | None      # explicit only; None if the field was absent
    has_completion_signal: bool   # derived per Ralph's precedence


def parse_status(text: str) -> LoopStatus:
    exit_signal = _to_bool(_EXIT_RE.search(text))
    status_m = _STATUS_RE.search(text)
    status = status_m.group(1).upper() if status_m else None
    progress = _to_bool(_PROGRESS_RE.search(text))
    summary_m = _SUMMARY_RE.search(text)
    summary = summary_m.group(1).strip() if summary_m else None

    # Precedence: explicit EXIT_SIGNAL is authoritative. Only when it is absent
    # does STATUS: COMPLETE infer a completion signal.
    if exit_signal is not None:
        has_completion_signal = exit_signal
    else:
        has_completion_signal = (status == "COMPLETE")

    return LoopStatus(summary=summary, progress=progress, status=status,
                      exit_signal=exit_signal,
                      has_completion_signal=has_completion_signal)
```

# src/mindhive/loop/budget.py
from dataclasses import dataclass


@dataclass
class BudgetEntry:
    ts: float       # epoch seconds
    tokens: int
    cost_usd: float


@dataclass
class BudgetConfig:
    window_hours: float
    token_budget: int        # 0 = unlimited
    cost_budget_usd: float   # 0 = unlimited


def _in_window(entries, now, window_hours):
    cutoff = now - window_hours * 3600.0
    return [e for e in entries if e.ts > cutoff]


def used(entries, now, window_hours):
    """Return (tokens, cost) summed over entries within the rolling window."""
    win = _in_window(entries, now, window_hours)
    return sum(e.tokens for e in win), sum(e.cost_usd for e in win)


def rolling_avg(entries, n):
    """Mean (tokens, cost) of the most recent n entries; (0,0) if empty."""
    recent = entries[-n:] if n > 0 else []
    if not recent:
        return 0, 0.0
    return (sum(e.tokens for e in recent) // len(recent),
            sum(e.cost_usd for e in recent) / len(recent))


def would_exceed(entries, now, cfg, next_tokens, next_cost):
    """True if adding (next_tokens, next_cost) would breach a non-zero budget."""
    used_t, used_c = used(entries, now, cfg.window_hours)
    if cfg.token_budget and used_t + next_tokens > cfg.token_budget:
        return True
    if cfg.cost_budget_usd and used_c + next_cost > cfg.cost_budget_usd:
        return True
    return False


def wait_seconds(entries, now, cfg, next_tokens, next_cost):
    """Seconds to wait until enough oldest entries age out of the window so the
    next iteration fits. 0.0 if it already fits. Drops entries oldest-first and
    returns the time at which the last entry that must leave exits the window."""
    if not would_exceed(entries, now, cfg, next_tokens, next_cost):
        return 0.0
    win = sorted(_in_window(entries, now, cfg.window_hours), key=lambda e: e.ts)
    used_t = sum(e.tokens for e in win)
    used_c = sum(e.cost_usd for e in win)
    wait = 0.0
    for e in win:
        used_t -= e.tokens
        used_c -= e.cost_usd
        # time when this entry leaves the window
        wait = (e.ts + cfg.window_hours * 3600.0) - now
        fits_t = (not cfg.token_budget) or (used_t + next_tokens <= cfg.token_budget)
        fits_c = (not cfg.cost_budget_usd) or (used_c + next_cost <= cfg.cost_budget_usd)
        if fits_t and fits_c:
            break
    return max(0.0, wait)
```

# src/mindhive/loop/context.py
from dataclasses import dataclass


@dataclass
class ContextConfig:
    context_window: int
    smart_zone_ratio: float


def occupancy_from(input_tokens: int, cache_read_input_tokens: int) -> int:
    """Proxy for current context size: the tokens the model read this turn."""
    return input_tokens + cache_read_input_tokens


def fill_ratio(occupancy: int, window: int) -> float:
    if window <= 0:
        return 0.0
    return occupancy / window


def needs_reset(occupancy: int, cfg: ContextConfig) -> bool:
    """True once the context leaves the smart zone (fill >= ratio), meaning the
    loop should end this Claude session and reseed from the compact briefing."""
    return fill_ratio(occupancy, cfg.context_window) >= cfg.smart_zone_ratio
```

# src/mindhive/loop/safeguards.py
from dataclasses import dataclass


# --- dual-condition exit gate ---
def should_exit(status, required_remaining: int, json_mode: bool):
    """Return (should_exit, reason). Stops only when BOTH hold:
      - explicit EXIT_SIGNAL is true (authoritative), AND
      - no required fix_plan items remain.
    Explicit EXIT_SIGNAL: false always continues. In JSON mode, a missing
    explicit signal can never trigger exit (keyword-only exit suppressed)."""
    if status.exit_signal is False:
        return False, "explicit_continue"
    explicit_done = status.exit_signal is True
    inferred_done = (not json_mode) and status.has_completion_signal
    if not (explicit_done or inferred_done):
        return False, "no_completion_signal"
    if required_remaining > 0:
        return False, "required_items_remain"
    return True, "completed"


# --- circuit breaker ---
@dataclass
class BreakerConfig:
    no_progress_threshold: int
    same_error_threshold: int
    cooldown_seconds: float


class CircuitBreaker:
    def __init__(self, cfg: BreakerConfig):
        self.cfg = cfg
        self.state = "CLOSED"
        self.consecutive_no_progress = 0
        self.consecutive_same_error = 0
        self.last_error_sig = None
        self.opened_at = None

    def record(self, progress: bool, error_sig, now: float) -> None:
        # no-progress tracking
        self.consecutive_no_progress = 0 if progress else self.consecutive_no_progress + 1
        # repeated-error tracking
        if error_sig and error_sig == self.last_error_sig:
            self.consecutive_same_error += 1
        elif error_sig:
            self.consecutive_same_error = 1
        else:
            self.consecutive_same_error = 0
        self.last_error_sig = error_sig

        if self.state == "HALF_OPEN":
            # trial iteration: progress closes, otherwise re-open
            if progress and not error_sig:
                self._close()
            else:
                self._open(now)
            return

        if (self.consecutive_no_progress >= self.cfg.no_progress_threshold or
                self.consecutive_same_error >= self.cfg.same_error_threshold):
            self._open(now)

    def allow(self, now: float) -> bool:
        """Whether the next iteration may run. Transitions OPEN -> HALF_OPEN once
        the cooldown has elapsed."""
        if self.state == "OPEN":
            if self.opened_at is not None and now - self.opened_at >= self.cfg.cooldown_seconds:
                self.state = "HALF_OPEN"
                return True
            return False
        return True

    def _open(self, now):
        self.state = "OPEN"
        self.opened_at = now

    def _close(self):
        self.state = "CLOSED"
        self.consecutive_no_progress = 0
        self.consecutive_same_error = 0
        self.opened_at = None
```

# src/mindhive/loop/driver.py
import json
import shutil
import subprocess
from dataclasses import dataclass, field


@dataclass
class Usage:
    input_tokens: int = 0
    output_tokens: int = 0
    cache_creation_input_tokens: int = 0
    cache_read_input_tokens: int = 0
    cost_usd: float = 0.0

    @property
    def total_tokens(self) -> int:
        return (self.input_tokens + self.output_tokens
                + self.cache_creation_input_tokens + self.cache_read_input_tokens)

    @property
    def context_occupancy(self) -> int:
        return self.input_tokens + self.cache_read_input_tokens


@dataclass
class DriverResult:
    text: str
    usage: Usage
    session_id: str | None
    num_turns: int
    rate_limited: bool
    raw: dict = field(default_factory=dict)


def preflight(cmd: str = "claude") -> bool:
    """True if the CLI is on PATH."""
    return shutil.which(cmd) is not None


def build_argv(cmd, prompt, loop_context, allowed_tools, output_format,
               session_id, model, effort):
    argv = [cmd, "--output-format", output_format]
    if model:
        argv += ["--model", model]
    if effort:
        argv += ["--effort", effort]
    if allowed_tools:
        argv += ["--allowedTools", allowed_tools]
    if session_id:
        # --resume with an explicit id, never --continue (which can hijack the
        # most-recent session in the directory).
        argv += ["--resume", session_id]
    argv += ["--append-system-prompt", loop_context]
    argv += ["-p", prompt]
    return argv


def parse_result(stdout: str) -> DriverResult:
    stdout = stdout.strip()
    try:
        obj = json.loads(stdout)
    except (json.JSONDecodeError, ValueError):
        return DriverResult(text=stdout, usage=Usage(), session_id=None,
                            num_turns=0, rate_limited=False, raw={})
    u = obj.get("usage", {}) or {}
    usage = Usage(
        input_tokens=u.get("input_tokens", 0),
        output_tokens=u.get("output_tokens", 0),
        cache_creation_input_tokens=u.get("cache_creation_input_tokens", 0),
        cache_read_input_tokens=u.get("cache_read_input_tokens", 0),
        cost_usd=obj.get("total_cost_usd", 0.0),
    )
    rate_limited = (
        "rate_limit_event" in obj
        and isinstance(obj.get("rate_limit_event"), dict)
        and obj["rate_limit_event"].get("status") == "rejected"
    )
    return DriverResult(
        text=obj.get("result", "") or "",
        usage=usage,
        session_id=obj.get("session_id"),
        num_turns=obj.get("num_turns", 0),
        rate_limited=rate_limited,
        raw=obj,
    )


class CLIDriver:
    """Real Claude Code CLI driver. Not exercised in unit tests; the engine
    accepts any object with a matching run() for fakes."""

    def __init__(self, cmd: str = "claude"):
        self.cmd = cmd

    def run(self, prompt, loop_context, *, allowed_tools, session_id,
            model, effort, timeout_s, live=False) -> DriverResult:
        output_format = "stream-json" if live else "json"
        argv = build_argv(self.cmd, prompt, loop_context, allowed_tools,
                          output_format, session_id, model, effort)
        proc = subprocess.run(
            argv, capture_output=True, text=True, timeout=timeout_s,
            creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0),
        )
        return parse_result(proc.stdout or proc.stderr)
```

# src/mindhive/loop/prompt.py

CONTRACT_INSTRUCTION = """\
When you finish this iteration, end your response with this exact block:

LOOP_STATUS:
  summary: <one line of what you did this iteration>
  progress: true|false
  STATUS: IN_PROGRESS|COMPLETE
  EXIT_SIGNAL: true|false

Set EXIT_SIGNAL: true ONLY when the ENTIRE task is finished and verified.
If more work remains, set EXIT_SIGNAL: false even if this iteration went well.
"""


def build_loop_context(briefing_text: str) -> str:
    """System-prompt context for an iteration: the compact MindHive briefing
    plus the standing completion contract."""
    return f"{briefing_text.rstrip()}\n\n---\n\n{CONTRACT_INSTRUCTION}"


def build_task_prompt(tasks, recall_hits) -> str:
    parts = []
    if tasks.prompt.strip():
        parts.append("# Goal\n" + tasks.prompt.strip())
    if tasks.fix_plan.strip():
        parts.append("# Fix plan (work the unchecked items, top priority first)\n"
                     + tasks.fix_plan.strip())
    if tasks.agent.strip():
        parts.append("# Build / test / run commands\n" + tasks.agent.strip())
    if recall_hits:
        lines = ["# Relevant prior memory"]
        for h in recall_hits:
            lines.append(f"- {h.get('snippet', '')}")
        parts.append("\n".join(lines))
    return "\n\n".join(parts) if parts else "Continue the task."

# src/mindhive/loop/engine.py
import time
from pathlib import Path

from .. import events, guard, recall as recall_mod
from ..briefing import build_briefing
from . import context as ctx_mod
from . import prompt as prompt_mod
from . import safeguards
from . import task as task_mod
from .driver import preflight


def run_loop(project, loop_dir, cfg, driver, *,
             now_fn=time.time, sleep_fn=time.sleep, progress_fn=None,
             live=False, force=False):
    """Drive one autonomous run to completion or a safeguard boundary.

    progress_fn(iteration_index)->bool is injected in tests to report whether
    files changed; in production it is a git/mtime diff (Task 11)."""
    loop_dir = Path(loop_dir)
    guard.ensure_safe(force=force)              # LIVE_LOCK check

    cb = safeguards.CircuitBreaker(safeguards.BreakerConfig(
        cfg.no_progress_threshold, cfg.same_error_threshold, cfg.cooldown_seconds))
    ccfg = ctx_mod.ContextConfig(cfg.context_window, cfg.smart_zone_ratio)

    events.append_event("loop_run_start",
                        {"goal_dir": str(loop_dir), "max_iterations": cfg.max_iterations},
                        project=project)

    start = now_fn()
    session_id = None
    iteration = 0
    reason = "max_iterations"

    while iteration < cfg.max_iterations:
        if now_fn() - start >= cfg.max_hours * 3600:
            reason = "max_hours"
            break
        if not cb.allow(now=now_fn()):
            reason = "breaker_open"
            break

        iteration += 1
        tasks = task_mod.load_tasks(loop_dir)
        briefing_text = build_briefing(events.read_all(project))
        hits = recall_mod.recall(tasks.prompt[:120], project=project, limit=3) if tasks.prompt else []
        loop_context = prompt_mod.build_loop_context(briefing_text)
        task_prompt = prompt_mod.build_task_prompt(tasks, hits)

        result = driver.run(
            task_prompt, loop_context,
            allowed_tools=cfg.allowed_tools, session_id=session_id,
            model=cfg.model, effort=cfg.effort,
            timeout_s=cfg.iteration_timeout_seconds, live=live)

        from .contract import parse_status
        status = parse_status(result.text)
        progressed = progress_fn(iteration) if progress_fn else True
        error_sig = "rate_limited" if result.rate_limited else None
        occupancy = result.usage.context_occupancy

        events.append_event("loop_iteration", {
            "iteration": iteration,
            "summary": status.summary or "",
            "progress": progressed,
            "cost_usd": result.usage.cost_usd,
            "total_tokens": result.usage.total_tokens,
            "context_fill_ratio": ctx_mod.fill_ratio(occupancy, cfg.context_window),
            "breaker_state": cb.state,
            "session_id": result.session_id,
        }, project=project)

        if result.rate_limited:
            reason = "api_cap"
            break

        required_remaining = task_mod.required_remaining_count(tasks.fix_plan)
        done, exit_reason = safeguards.should_exit(status, required_remaining, json_mode=not live)
        if done:
            reason = "completed"
            break

        cb.record(progress=progressed, error_sig=error_sig, now=now_fn())
        if cb.state == "OPEN":
            reason = "breaker_open"
            break

        # context smart-zone: keep session warm, or reset before the dumb zone
        if ctx_mod.needs_reset(occupancy, ccfg):
            events.append_event("compaction",
                                {"text": f"loop context reset at fill "
                                         f"{ctx_mod.fill_ratio(occupancy, cfg.context_window):.2f}"},
                                project=project)
            session_id = None
        else:
            session_id = result.session_id

    events.append_event("loop_run_end",
                        {"reason": reason, "iterations": iteration}, project=project)
    return {"reason": reason, "iterations": iteration}

# src/mindhive/loop/progress.py
from pathlib import Path

_IGNORE = {".git", ".hive-loop", "__pycache__", ".venv", "node_modules"}


class MtimeTracker:
    """Cheap, dependency-free progress detector: snapshots the (path, mtime,
    size) of project files and reports whether anything changed since."""

    def __init__(self, root: Path):
        self.root = Path(root)
        self._prev = {}

    def _scan(self):
        state = {}
        for p in self.root.rglob("*"):
            if any(part in _IGNORE for part in p.parts):
                continue
            if p.is_file():
                st = p.stat()
                state[str(p)] = (st.st_mtime, st.st_size)
        return state

    def snapshot(self):
        self._prev = self._scan()

    def changed(self) -> bool:
        current = self._scan()
        changed = current != self._prev
        self._prev = current
        return changed
```

# src/mindhive/loop/state.py
from .. import events


def last_run_status(project=None):
    """Summarize the most recent loop run from the event ledger, or None."""
    evs = events.read_all(project)
    starts = [e for e in evs if e.type == "loop_run_start"]
    if not starts:
        return None
    last_start_ts = starts[-1].ts
    iters, ends = [], []
    for e in evs:
        if e.ts < last_start_ts:
            continue
        if e.type == "loop_iteration":
            iters.append(e)
        elif e.type == "loop_run_end":
            ends.append(e)
    end = ends[-1] if ends else None
    return {
        "started": last_start_ts,
        "reason": (end.payload.get("reason") if end else "running"),
        "iterations": (end.payload.get("iterations") if end else len(iters)),
        "total_cost_usd": sum(e.payload.get("cost_usd", 0.0) for e in iters),
        "total_tokens": sum(e.payload.get("total_tokens", 0) for e in iters),
    }
```
